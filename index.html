using BestInScript.API.Services; using Microsoft.OpenApi.Models; var builder = WebApplication.CreateBuilder(args); // ── Services ─────────────────────────────────────────────────────────────────── builder.Services.AddControllers(); // Swagger / OpenAPI builder.Services.AddEndpointsApiExplorer(); builder.Services.AddSwaggerGen(c => { c.SwaggerDoc("v1", new OpenApiInfo { Title = "BestInScript API", Version = "v1", Description = "ARPG macro engine – assign keyboard scripts to toggle keys." }); }); // App services (singletons so the same instance is shared) builder.Services.AddSingleton (); builder.Services.AddSingleton (); builder.Services.AddSingleton (); // Register HotkeyEngine as IHostedService so StartAsync/StopAsync are called builder.Services.AddHostedService(sp => sp.GetRequiredService ()); var app = builder.Build(); // ── Middleware ──────────────────────────────────────────────────────────────── app.UseSwagger(); app.UseSwaggerUI(c => { c.SwaggerEndpoint("/swagger/v1/swagger.json", "BestInScript v1"); c.RoutePrefix = "swagger"; }); // Resolve wwwroot relative to the project source directory so it works whether // the app is launched via "dotnet run" (CWD = project folder) or from bin/publish. // ContentRootPath is set by ASP.NET Core to the project directory when using // "dotnet run", and to the publish output directory when published — both cases // have wwwroot next to them, so this is always correct. var wwwrootPath = Path.Combine(builder.Environment.ContentRootPath, "wwwroot"); var fileProvider = new Microsoft.Extensions.FileProviders.PhysicalFileProvider(wwwrootPath); app.UseDefaultFiles(new DefaultFilesOptions { FileProvider = fileProvider, RequestPath = "" }); app.UseStaticFiles(new StaticFileOptions { FileProvider = fileProvider, RequestPath = "" }); app.UseAuthorization(); app.MapControllers(); app.Run();
